cmake_minimum_required(VERSION 3.16)
project(bs0013_week09_bst LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UndefinedBehaviorSanitizer" OFF)
enable_testing()
add_library(int_bst src/int_bst.cpp)
target_include_directories(int_bst PUBLIC include)
target_compile_options(int_bst PRIVATE -Wall -Wextra -Wpedantic)
add_executable(week09_tests tests/test_int_bst.cpp)
target_link_libraries(week09_tests PRIVATE int_bst)
target_compile_options(week09_tests PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(int_bst PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(int_bst PRIVATE -fsanitize=address,undefined)
    target_compile_options(week09_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(week09_tests PRIVATE -fsanitize=address,undefined)
endif()
add_test(NAME week09_tests COMMAND week09_tests)
