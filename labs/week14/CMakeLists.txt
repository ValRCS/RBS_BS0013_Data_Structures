cmake_minimum_required(VERSION 3.16)
project(bs0013_week14_indexes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
add_library(indexes src/indexes.cpp)
target_include_directories(indexes PUBLIC include)
add_executable(week14_tests tests/test_indexes.cpp)
add_executable(week14_benchmark src/benchmark.cpp)
foreach(target week14_tests week14_benchmark)
  target_link_libraries(${target} PRIVATE indexes)
endforeach()
foreach(target indexes week14_tests week14_benchmark)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
    if(ENABLE_SANITIZERS)
      target_compile_options(${target} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
      target_link_options(${target} PRIVATE -fsanitize=address,undefined)
    endif()
  endif()
endforeach()
enable_testing()
add_test(NAME week14_indexes_public COMMAND week14_tests)
