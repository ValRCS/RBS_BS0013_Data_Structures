cmake_minimum_required(VERSION 3.16)
project(bs0013_week13_btree LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable AddressSanitizer and UBSan" OFF)
add_library(btree src/btree.cpp)
target_include_directories(btree PUBLIC include)
add_executable(week13_tests tests/test_btree.cpp)
target_link_libraries(week13_tests PRIVATE btree)
target_compile_options(btree PRIVATE -Wall -Wextra -Wpedantic)
target_compile_options(week13_tests PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(btree PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(btree PRIVATE -fsanitize=address,undefined)
 target_compile_options(week13_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
 target_link_options(week13_tests PRIVATE -fsanitize=address,undefined)
endif()
enable_testing()
add_test(NAME week13_btree_public COMMAND week13_tests)
