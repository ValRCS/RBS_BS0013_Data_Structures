cmake_minimum_required(VERSION 3.16)
project(bs0013_week11_graphs LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
add_library(graph src/graph.cpp)
target_include_directories(graph PUBLIC include)
add_executable(week11_tests tests/test_graph.cpp)
target_link_libraries(week11_tests PRIVATE graph)
foreach(t graph week11_tests)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME week11_graph_public COMMAND week11_tests)
