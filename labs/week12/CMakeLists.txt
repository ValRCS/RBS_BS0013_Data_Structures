cmake_minimum_required(VERSION 3.16)
project(bs0013_week12_trie LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable ASan and UBSan" OFF)
add_library(trie src/trie.cpp)
target_include_directories(trie PUBLIC include)
add_executable(week12_tests tests/test_trie.cpp)
target_link_libraries(week12_tests PRIVATE trie)
foreach(t trie week12_tests)
  target_compile_options(${t} PRIVATE -Wall -Wextra -Wpedantic)
  if(ENABLE_SANITIZERS AND CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(${t} PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(${t} PRIVATE -fsanitize=address,undefined)
  endif()
endforeach()
enable_testing()
add_test(NAME week12_trie_public COMMAND week12_tests)
