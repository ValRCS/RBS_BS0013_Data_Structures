cmake_minimum_required(VERSION 3.16)
project(bs0013_week10_avl LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(ENABLE_SANITIZERS "Enable sanitizers" OFF)
add_library(int_avl src/int_avl.cpp)
target_include_directories(int_avl PUBLIC include)
target_compile_options(int_avl PRIVATE -Wall -Wextra -Wpedantic)
add_executable(week10_tests tests/test_int_avl.cpp)
target_link_libraries(week10_tests PRIVATE int_avl)
target_compile_options(week10_tests PRIVATE -Wall -Wextra -Wpedantic)
if(ENABLE_SANITIZERS)
  if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    target_compile_options(int_avl PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_compile_options(week10_tests PRIVATE -fsanitize=address,undefined -fno-omit-frame-pointer)
    target_link_options(week10_tests PRIVATE -fsanitize=address,undefined)
  else()
    message(FATAL_ERROR "Sanitizers require GCC or Clang")
  endif()
endif()
enable_testing()
add_test(NAME week10_avl_public COMMAND week10_tests)
